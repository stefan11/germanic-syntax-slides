\subsection{Aufgaben / Übungen}


\frame{
\frametitle{Aufgaben}

\begin{enumerate}
\item Wiederholen Sie den Stoff im Buch.
\item Beantworten Sie die Quick questions im Buch und lesen Sie nach, wenn Sie dazu nicht in der
  Lage sein sollten.
\item Sollte etwas unverständlich sein, fragen Sie bei den Tutor*innen oder bei mir per Mail nach.

\item Lesen Sie in der aktuellen Duden-Grammatik den Abschnitt zu Valenz \citep[§1179--1182]{Duden2005}

\end{enumerate}

}

\frame{
\frametitle{Übungen}

\begin{itemize}
\item Lösen Sie die Übungen im Lehrbuch. 
\item Die Übungen entsprechen dem, was in der Klausur von Ihnen gefordert
wird. 
\item Die Lösungen sind am Ende des Buches zu finden.
\item Besuchen Sie das Tutorium zur Vertiefung und Übung der in der Vorlesung besprochenen Themen.

\item Sollte etwas unklar sein, schreiben Sie mir eine Mail.
\end{itemize}


}
